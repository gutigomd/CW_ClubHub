import { HttpInterceptorFn } from '@angular/common/http';
import { inject } from '@angular/core';
import { AuthService } from '../services/auth.service';

export const authInterceptor: HttpInterceptorFn = (req, next) => {
  const token = inject(AuthService).getToken();
  if (token) {
    req = req.clone({
      setHeaders: { Authorization: `Bearer ${token}` }
    });
  }
  return next(req);
};

import { Injectable } from '@angular/core';
import { HttpClient, HttpParams } from '@angular/common/http';
import { Community, ClubEvent, MyRegistration, CommunityPost } from '../models/api.models';

@Injectable({ providedIn: 'root' })
export class ApiService {
  constructor(private http: HttpClient) {}

  getCommunities(search?: string, category?: string, isPrivate?: string) {
    let params = new HttpParams();
    if (search) params = params.set('search', search);
    if (category) params = params.set('category', category);
    if (isPrivate === 'true' || isPrivate === 'false') params = params.set('isPrivate', isPrivate);
    return this.http.get<Community[]>('/api/communities', { params });
  }

  getCommunity(id: number) {
    return this.http.get<Community>(`/api/communities/${id}`);
  }

  createCommunity(data: { name: string; category: string; description: string; isPrivate: boolean }) {
    return this.http.post('/api/communities', data);
  }

  joinCommunity(id: number, applicationNote?: string) {
    return this.http.post<{ status: string; message: string }>(`/api/communities/${id}/join`, { applicationNote });
  }

  leaveCommunity(id: number) {
    return this.http.post<{ message: string }>(`/api/communities/${id}/leave`, {});
  }

  reviewMembership(communityId: number, membershipId: number, approve: boolean) {
    return this.http.post(`/api/communities/${communityId}/memberships/${membershipId}/review`, { approve });
  }

  revokeMembership(communityId: number, membershipId: number) {
    return this.http.post<{ message: string }>(`/api/communities/${communityId}/memberships/${membershipId}/revoke`, {});
  }

  toggleBlockCommunity(id: number, block: boolean) {
    return this.http.post(`/api/communities/${id}/block`, block);
  }

  getEvents(communityId?: number) {
    let params = new HttpParams();
    if (communityId) params = params.set('communityId', communityId);
    return this.http.get<ClubEvent[]>('/api/events', { params });
  }

  getMyRegistrations() {
    return this.http.get<MyRegistration[]>('/api/events/my');
  }

  createEvent(data: {
    communityId: number;
    title: string;
    description: string;
    location: string;
    startsAt: string;
    maxCapacity: number;
    isPrivateForMembersOnly: boolean;
  }) {
    return this.http.post('/api/events', data);
  }

  registerForEvent(eventId: number) {
    return this.http.post<{ status: string; message: string }>(`/api/events/${eventId}/register`, {});
  }

  cancelRegistration(eventId: number) {
    return this.http.post<{ message: string }>(`/api/events/${eventId}/cancel`, {});
  }

  getCommunityPosts(communityId: number) {
    return this.http.get<CommunityPost[]>(`/api/posts/community/${communityId}`);
  }

  createPost(data: { communityId: number; title: string; content: string; isMembersOnly: boolean }) {
    return this.http.post('/api/posts', data);
  }

  moderatePost(postId: number, hide: boolean, reason?: string) {
    return this.http.post(`/api/posts/${postId}/moderate`, { hide, reason });
  }
}

  login(email: string, password: string) {
    return this.http.post<AuthResponse>('/api/auth/login', { email, password }).pipe(
      tap(res => this.saveAuth(res))
    );
  }

  register(fullName: string, email: string, password: string, role: number) {
    return this.http.post<AuthResponse>('/api/auth/register', { fullName, email, password, role }).pipe(
      tap(res => this.saveAuth(res))
    );
  }

  logout() {
    localStorage.removeItem(this.storageKey);
    this.userSignal.set(null);
  }

  getToken(): string | null {
    return this.userSignal()?.token ?? null;
  }

  private saveAuth(res: AuthResponse) {
    localStorage.setItem(this.storageKey, JSON.stringify(res));
    this.userSignal.set(res);
  }

  private loadFromStorage(): AuthResponse | null {
    try {
      const raw = localStorage.getItem(this.storageKey);
      return raw ? JSON.parse(raw) : null;
    } catch {
      return null;
    }
  }
}

[ApiController]
[Route("api/[controller]")]
public class AuthController : ControllerBase
{
    private readonly AppDbContext _db;
    private readonly IConfiguration _config;

    public AuthController(AppDbContext db, IConfiguration config)
    {
        _db = db;
        _config = config;
    }

    [HttpPost("register")]
    public async Task<ActionResult<AuthResponse>> Register(RegisterRequest req)
    {
        if (await _db.Users.AnyAsync(u => u.Email.ToLower() == req.Email.ToLower()))
            return Conflict(new { message = "Користувач із такою електронною поштою вже існує." });

        // Забороняємо самостійну реєстрацію модератора через публічну форму
        var assignedRole = req.Role == UserRole.Moderator ? UserRole.Participant : req.Role;

        var user = new User
        {
            FullName = req.FullName.Trim(),
            Email = req.Email.Trim().ToLower(),
            PasswordHash = PasswordHasher.Hash(req.Password),
            Role = assignedRole
        };

        _db.Users.Add(user);
        await _db.SaveChangesAsync();

        return Ok(BuildAuthResponse(user));
    }

    [HttpPost("login")]
    public async Task<ActionResult<AuthResponse>> Login(LoginRequest req)
    {
        var user = await _db.Users.FirstOrDefaultAsync(u => u.Email.ToLower() == req.Email.Trim().ToLower());
        if (user == null || !PasswordHasher.Verify(req.Password, user.PasswordHash))
            return Unauthorized(new { message = "Невірний email або пароль." });

        return Ok(BuildAuthResponse(user));
    }

    private AuthResponse BuildAuthResponse(User user)
    {
        var claims = new[]
        {
            new Claim(ClaimTypes.NameIdentifier, user.Id.ToString()),
            new Claim(ClaimTypes.Email, user.Email),
            new Claim(ClaimTypes.Name, user.FullName),
            new Claim(ClaimTypes.Role, user.Role.ToString())
        };

        var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes(_config["Jwt:Key"]!));
        var creds = new SigningCredentials(key, SecurityAlgorithms.HmacSha256);
        var expires = DateTime.UtcNow.AddHours(int.Parse(_config["Jwt:ExpireHours"] ?? "12"));

        var token = new JwtSecurityToken(
            issuer: _config["Jwt:Issuer"],
            audience: _config["Jwt:Audience"],
            claims: claims,
            expires: expires,
            signingCredentials: creds
        );

        return new AuthResponse(
            new JwtSecurityTokenHandler().WriteToken(token),
            user.Id,
            user.FullName,
            user.Email,
            user.Role.ToString()
        );
    }
}

    private async Task RevokeMembershipAndCascadeEvents(Membership membership)
    {
        using var tx = await _db.Database.BeginTransactionAsync();

        membership.Status = MembershipStatus.Revoked;
        membership.UpdatedAt = DateTime.UtcNow;

            .Where(r => r.UserId == membership.UserId
                        && r.Event!.CommunityId == membership.CommunityId
                        && r.Event.StartsAt > DateTime.UtcNow
                        && r.Status != RegistrationStatus.Cancelled)
            .ToListAsync();

        foreach (var reg in futureRegs)
        {
            var wasConfirmed = reg.Status == RegistrationStatus.Confirmed;
            reg.Status = RegistrationStatus.Cancelled;
            reg.UpdatedAt = DateTime.UtcNow;

                {
                    nextInWaitlist.Status = RegistrationStatus.Confirmed;
                    nextInWaitlist.UpdatedAt = DateTime.UtcNow;
                }
            }
        }

        await _db.SaveChangesAsync();
        await tx.CommitAsync();
    }

        var msg = assignedStatus == RegistrationStatus.Confirmed
            ? "Ви успішно зареєстровані на подію!"
            : "Вільних місць немає. Вас додано до списку очікування (Waitlist).";

        return Ok(new { status = assignedStatus.ToString(), message = msg });
    }

    // Скасування участі з автоматичним переведенням першого зі списку очікування
    [Authorize]
    [HttpPost("{id:int}/cancel")]
    public async Task<IActionResult> CancelRegistration(int id)
    {
        var userId = CurrentUserId!.Value;

        using var tx = await _db.Database.BeginTransactionAsync();

        var reg = await _db.EventRegistrations
            .FirstOrDefaultAsync(r => r.EventId == id && r.UserId == userId && r.Status != RegistrationStatus.Cancelled);

        if (reg == null)
            return NotFound(new { message = "Активну реєстрацію не знайдено." });

        var wasConfirmed = reg.Status == RegistrationStatus.Confirmed;
        reg.Status = RegistrationStatus.Cancelled;
        reg.UpdatedAt = DateTime.UtcNow;

        bool promotedSomeone = false;
        if (wasConfirmed)
        {
            var nextInLine = await _db.EventRegistrations
                .Where(r => r.EventId == id && r.Status == RegistrationStatus.Waitlisted)
                .OrderBy(r => r.RegisteredAt)
                .FirstOrDefaultAsync();

            if (nextInLine != null)
            {
                nextInLine.Status = RegistrationStatus.Confirmed;
                nextInLine.UpdatedAt = DateTime.UtcNow;
                promotedSomeone = true;
            }
        }

        await _db.SaveChangesAsync();
        await tx.CommitAsync();

        return Ok(new
        {
            message = promotedSomeone
                ? "Реєстрацію скасовано. Звільнене місце автоматично передано першому учаснику зі списку очікування."
                : "Вашу реєстрацію скасовано."
        });
    }

    [HttpGet("community/{communityId:int}")]
    public async Task<IActionResult> GetCommunityPosts(int communityId)
    {
        var userId = CurrentUserId;
        var isModerator = CurrentUserRole == nameof(UserRole.Moderator);

        var community = await _db.Communities.FindAsync(communityId);
        if (community == null || (community.IsBlocked && !isModerator))
            return NotFound(new { message = "Спільноту не знайдено." });

        var isOrganizer = userId.HasValue && community.OrganizerId == userId.Value;
        var isApprovedMember = userId.HasValue && await _db.Memberships.AnyAsync(m =>
            m.CommunityId == communityId && m.UserId == userId.Value && m.Status == MembershipStatus.Approved);

        var hasPrivateAccess = isOrganizer || isApprovedMember || isModerator;

        // Якщо спільнота закрита і користувач не учасник — жорстка відмова 403
        if (community.IsPrivate && !hasPrivateAccess)
            return StatusCode(403, new { message = "Доступ заборонено. Це закрита спільнота лише для чинних учасників." });

            .Include(p => p.Author)

            query = query.Where(p => !p.IsMembersOnly);

        var posts = await query.OrderByDescending(p => p.CreatedAt).Select(p => new
        {
            p.Id,
            p.Title,
            p.Content,
            p.IsMembersOnly,
            p.IsHiddenByModerator,
            p.ModerationReason,
            AuthorName = p.Author!.FullName,
            p.CreatedAt
        }).ToListAsync();

        return Ok(posts);
    }

    {

        var isModerator = CurrentUserRole == nameof(UserRole.Moderator);

        var post = await _db.Posts
            .Include(p => p.Community)
            .Include(p => p.Author)
            .FirstOrDefaultAsync(p => p.Id == id);

        if (post == null || (post.IsHiddenByModerator && !isModerator))
            return NotFound(new { message = "Публікацію не знайдено." });

        if (post.Community!.IsPrivate || post.IsMembersOnly)
        {
            var isOrganizer = userId.HasValue && post.Community.OrganizerId == userId.Value;
            var isMember = userId.HasValue && await _db.Memberships.AnyAsync(m =>
                m.CommunityId == post.CommunityId && m.UserId == userId.Value && m.Status == MembershipStatus.Approved);

            if (!isOrganizer && !isMember && !isModerator)
                return StatusCode(403, new { message = "Доступ до закритої публікації заборонено." });
        }

        return Ok(new
        {
            post.Id,
            post.CommunityId,
            post.Title,
            post.Content,
            post.IsMembersOnly,
            AuthorName = post.Author?.FullName,
            post.CreatedAt
        });
    }

            m.CommunityId == req.CommunityId && m.UserId == userId && m.Status == MembershipStatus.Approved);

using System.Security.Cryptography;

namespace Server.Services;

public static class PasswordHasher
{
    public static string Hash(string password)
    {
        byte[] salt = RandomNumberGenerator.GetBytes(16);
        byte[] hash = Rfc2898DeriveBytes.Pbkdf2(password, salt, 100_000, HashAlgorithmName.SHA256, 32);
        return $"{Convert.ToBase64String(salt)}.{Convert.ToBase64String(hash)}";
    }

    public static bool Verify(string password, string storedHash)
    {
        var parts = storedHash.Split('.');
        if (parts.Length != 2) return false;
        byte[] salt = Convert.FromBase64String(parts[0]);
        byte[] expected = Convert.FromBase64String(parts[1]);
        byte[] actual = Rfc2898DeriveBytes.Pbkdf2(password, salt, 100_000, HashAlgorithmName.SHA256, 32);
        return CryptographicOperations.FixedTimeEquals(actual, expected);
    }
}
